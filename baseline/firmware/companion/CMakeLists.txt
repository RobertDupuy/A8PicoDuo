cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(a8duo_library_b C CXX ASM)
set(PICO_BOARD_HEADER_DIRS ${CMAKE_CURRENT_SOURCE_DIR})
set(PICO_BOARD myboard)
pico_sdk_init()
set(E ${CMAKE_CURRENT_LIST_DIR}/../engine)
add_executable(a8duo_library_b main.c ../../game/game_engine.c msc_disk.c usb_descriptors.c ${E}/flash_fs.c ${E}/fatfs_disk.c ${E}/fatfs/diskio.c ${E}/fatfs/ff.c ${E}/fatfs/ffunicode.c)
target_include_directories(a8duo_library_b PRIVATE . ${E} ${E}/fatfs ../shared)
target_compile_definitions(a8duo_library_b PRIVATE DUO_COMPANION=1 PICO_FLASH_SIZE_BYTES=16777216)
target_link_libraries(a8duo_library_b pico_stdlib hardware_uart hardware_flash tinyusb_device)
pico_enable_stdio_uart(a8duo_library_b 0)
pico_enable_stdio_usb(a8duo_library_b 0)
pico_add_extra_outputs(a8duo_library_b)
# Optional, clearly named recovery tool. It erases only the module receiving it.
add_executable(a8duo_ERASE_LIBRARY erase_data.c)
target_compile_definitions(a8duo_ERASE_LIBRARY PRIVATE PICO_FLASH_SIZE_BYTES=16777216)
target_link_libraries(a8duo_ERASE_LIBRARY pico_stdlib hardware_flash)
pico_enable_stdio_uart(a8duo_ERASE_LIBRARY 0)
pico_enable_stdio_usb(a8duo_ERASE_LIBRARY 0)
pico_add_extra_outputs(a8duo_ERASE_LIBRARY)

set_target_properties(a8duo_library_b a8duo_ERASE_LIBRARY PROPERTIES LINKER_LANGUAGE C)
